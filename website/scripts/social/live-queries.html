const Todo = Ramose.Entity("todo", {
  title: Ramose.string(),
  done: Ramose.boolean(),
});

const todos = Ramose.Query.from(Todo);

const Todos = () => {
  const { data } = useLiveQuery(db, todos);
  return (
    <ul>
      {data?.map((todo) => (
        <li key={todo.id}>{todo.title}</li>
      ))}
    </ul>
  );
};
